```python
from moviepy.editor import VideoFileClip, clips_array, AudioFileClip

class VideoEditor:
    """Handles video editing tasks."""
    
    @staticmethod
    def create_split_screen(video_path, image_path, output_path):
        """Creates a split-screen video with gameplay on one side and an image on the other."""
        video_clip = VideoFileClip(video_path)
        image_clip = ImageClip(image_path).set_duration(video_clip.duration)
        
        final_clip = clips_array([[video_clip, image_clip]])
        final_clip.write_videofile(output_path, codec="libx264", fps=24)
    
    @staticmethod
    def add_background_music(video_path, music_path, output_path):
        """Adds background music to the video."""
        video_clip = VideoFileClip(video_path)
        audio_clip = AudioFileClip(music_path)
        
        final_clip = video_clip.set_audio(audio_clip)
        final_clip.write_videofile(output_path, codec="libx264", fps=24)
```
